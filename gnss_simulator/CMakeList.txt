cmake_minimum_required(VERSION 3.20)
project(gr-gnss-sim VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Поиск зависимостей
find_package(gnuradio REQUIRED)
find_package(fmt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SOAPYSDR REQUIRED SoapySDR)

# Основная библиотека
add_library(gnss-sim INTERFACE)
target_include_directories(gnss-sim INTERFACE
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
    ${SOAPYSDR_INCLUDE_DIRS}
)
target_link_libraries(gnss-sim INTERFACE
    gnuradio::core
    fmt::fmt
    ${SOAPYSDR_LIBRARIES}
)

# Hello World (файловый вывод)
add_executable(gnss_hello_world apps/gnss_hello_world.cpp)
target_link_libraries(gnss_hello_world PRIVATE gnss-sim)

# HackRF output
add_executable(gnss_hackrf_output apps/gnss_hackrf_output.cpp)
target_link_libraries(gnss_hackrf_output PRIVATE gnss-sim)
